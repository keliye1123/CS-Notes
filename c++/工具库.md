## 一、json解析库

json解析库在使用时要配合try catch使用

~~~cpp
#include <fstream>
#include <iostream>
#include "spdlog/spdlog.h"
#include "nlohmann/json.hpp"

int main() {
    try {

        // nlohmann::ordered_json:按顺序保存
        std::ifstream input_file("../assets/json_example.json");
        nlohmann::json json_data = nlohmann::json::parse(input_file);
        input_file.close();
        spdlog::info("json加载成功");

        //中括号获取：当键不存在时值为空不出现异常
        std::string name = json_data["name"].get<std::string>();

        //.at获取：当键不存在时会抛出异常
        std::string name1 = json_data.at("name").get<std::string>();

        //直接深层嵌套对象获取
        double res = json_data["metadata"]["version"].get<double>();

        //安全访问先判断键是否存在，推荐访问方法
        if (json_data.contains("name1")) {
            spdlog::info("存在name1");
        }
        else {
            spdlog::info("不存在name1");
        }

        spdlog::info("{}",name);

        //判断空
        if (json_data["midd"].is_null()) {
            spdlog::info("mid为空");
        }

        //.value:当第一格参数的键不存在时，使用第二个参数
        int age = json_data.value("age1",10);
        spdlog::info("age:{}",age);

        //对象嵌套:创建nlohmann::json接受对象
        nlohmann::json address = json_data.at("address");
        std::string street = address.at("street").get<std::string>();

        //数组数据
        nlohmann::json hobbies = json_data.at("hobbies");
        for (auto it : hobbies) {
            spdlog::error("hoobies:{}",it.get<std::string>());
        }

        //保存
        std::ofstream out_file("../assets/json_example.json");
        out_file << json_data.dump(4);  //使用缩颈为4保存
        out_file.close();


        //获取数据类型和实际类型不匹配在catch中会报错
        int email = json_data["email"].get<int>();

    }catch (const std::exception& e) {
        spdlog::error("ERROR:{}",e.what());
    }
    return 0;
}

~~~
